    const AudioEngine = {
      enabled: true,
      ready: false,
      init() {
        if (this.ready) return;
        try {
          Tone.start();

          // Low-end leather impact
          this.leatherBass = new Tone.MembraneSynth({
            pitchDecay: 0.05,
            octaves: 4,
            oscillator: { type: "sine" },
            envelope: { attack: 0.001, decay: 0.15, sustain: 0, release: 0.08 }
          }).toDestination();
          this.leatherBass.volume.value = -3;

          // High-frequency punch snap/crack
          this.snapNoise = new Tone.NoiseSynth({
            noise: { type: "white" },
            envelope: { attack: 0.002, decay: 0.08, sustain: 0, release: 0.05 }
          }).toDestination();
          this.snapNoise.volume.value = -8;

          // Glove swoosh
          this.swish = new Tone.Filter({
            frequency: 1800,
            type: "bandpass",
            Q: 2.0
          }).toDestination();
          this.swishNoise = new Tone.NoiseSynth({
            noise: { type: "pink" },
            envelope: { attack: 0.01, decay: 0.1, sustain: 0, release: 0.05 }
          }).connect(this.swish);
          this.swishNoise.volume.value = -12;

          // Gacha Revelation Fanfare
          this.fanfare = new Tone.PolySynth(Tone.Synth, {
            oscillator: { type: "triangle8" },
            envelope: { attack: 0.01, decay: 1.0, sustain: 0.1, release: 0.8 }
          }).toDestination();
          this.fanfare.volume.value = -6;

          // Sub Knockout Explosion
          this.koBoom = new Tone.MembraneSynth({
            pitchDecay: 0.2,
            octaves: 6,
            oscillator: { type: "triangle" },
            envelope: { attack: 0.001, decay: 0.8, sustain: 0.05, release: 0.6 }
          }).toDestination();
          this.koBoom.volume.value = 2;

          this.ready = true;
        } catch (e) {
          console.warn("Audio Context waiting for gesture", e);
        }
      },
      playPunch(speedFactor = 1) {
        if (!this.enabled) return;
        this.init();
        Tone.start();

        const now = Tone.now();
        const baseNote = speedFactor > 2.5 ? "G2" : (speedFactor > 1.5 ? "F2" : "D2");
        this.leatherBass.triggerAttackRelease(baseNote, "16n", now);
        this.snapNoise.triggerAttackRelease("32n", now + 0.01);
      },
      playSwoosh() {
        if (!this.enabled) return;
        this.init();
        this.swishNoise.triggerAttackRelease("16n");
      },
      playKO() {
        if (!this.enabled) return;
        this.init();
        const now = Tone.now();
        this.koBoom.triggerAttackRelease("C1", "4n", now);
        this.snapNoise.triggerAttackRelease("8n", now);
        setTimeout(() => {
          this.fanfare.triggerAttackRelease(["E5", "G#5", "B5", "E6"], "2n");
        }, 350);
      }
    };

    function createPunchingBagTexture() {
      const canvas = document.createElement("canvas");
      canvas.width = 2048;
      canvas.height = 2048;
      const ctx = canvas.getContext("2d");

      // Distressed Tan Leather Gradient
      const leatherGrad = ctx.createLinearGradient(0, 0, 2048, 0);
      leatherGrad.addColorStop(0, "#854d0e");
      leatherGrad.addColorStop(0.25, "#b45309");
      leatherGrad.addColorStop(0.5, "#d97706");
      leatherGrad.addColorStop(0.75, "#b45309");
      leatherGrad.addColorStop(1, "#78350f");
      ctx.fillStyle = leatherGrad;
      ctx.fillRect(0, 0, 2048, 2048);

      // Vertical leather panel seams
      for (let x = 0; x <= 2048; x += 512) {
        ctx.strokeStyle = "rgba(0, 0, 0, 0.6)";
        ctx.lineWidth = 14;
        ctx.beginPath();
        ctx.moveTo(x, 0);
        ctx.lineTo(x, 2048);
        ctx.stroke();

        // Stitches
        ctx.strokeStyle = "rgba(255, 237, 213, 0.45)";
        ctx.lineWidth = 4;
        ctx.setLineDash([12, 12]);
        ctx.beginPath();
        ctx.moveTo(x - 12, 0);
        ctx.lineTo(x - 12, 2048);
        ctx.moveTo(x + 12, 0);
        ctx.lineTo(x + 12, 2048);
        ctx.stroke();
        ctx.setLineDash([]);
      }

      // Heavy Reinforced Black Leather Bottom Cap
      ctx.fillStyle = "#18181b";
      ctx.fillRect(0, 1600, 2048, 448);
      ctx.strokeStyle = "#000000";
      ctx.lineWidth = 18;
      ctx.strokeRect(0, 1600, 2048, 448);

      ctx.fillStyle = "#18181b";
      ctx.fillRect(0, 0, 2048, 240);
      ctx.strokeRect(0, 0, 2048, 240);

    // Concrete Gym Floor with Boxing Ring Chalk
    function createGymFloorTexture() {
      const canvas = document.createElement("canvas");
      canvas.width = 1024;
      canvas.height = 1024;
      const ctx = canvas.getContext("2d");

      ctx.fillStyle = "#12131a";
      ctx.fillRect(0, 0, 1024, 1024);

      for (let i = 0; i < 4000; i++) {
        ctx.fillStyle = Math.random() > 0.5 ? "rgba(255,255,255,0.02)" : "rgba(0,0,0,0.2)";
        ctx.fillRect(Math.random() * 1024, Math.random() * 1024, 4, 4);
      }

      // Spotlight ring chalk mark
      ctx.strokeStyle = "rgba(234, 179, 8, 0.25)";
      ctx.lineWidth = 8;
      ctx.beginPath();
      ctx.arc(512, 512, 380, 0, Math.PI * 2);
      ctx.stroke();

      ctx.strokeStyle = "rgba(255, 255, 255, 0.15)";
      ctx.lineWidth = 4;
      ctx.beginPath();
      ctx.arc(512, 512, 280, 0, Math.PI * 2);
      ctx.stroke();

      const texture = new THREE.CanvasTexture(canvas);
      texture.wrapS = THREE.RepeatWrapping;
      texture.wrapT = THREE.RepeatWrapping;
      return texture;
    }

    // Anime outline generator using inverted hull
    function createAnimeOutline(geometry, thickness = 0.05) {
      const outlineGeo = geometry.clone();
      const pos = outlineGeo.attributes.position;
      const norm = outlineGeo.attributes.normal;

      if (norm) {
        for (let i = 0; i < pos.count; i++) {
          pos.setXYZ(
            i,
            pos.getX(i) + norm.getX(i) * thickness,
            pos.getY(i) + norm.getY(i) * thickness,
            pos.getZ(i) + norm.getZ(i) * thickness
          );
        }
        outlineGeo.computeVertexNormals();
      }

      const outlineMat = new THREE.MeshBasicMaterial({
        color: 0x000000,
        side: THREE.BackSide
      });
      return new THREE.Mesh(outlineGeo, outlineMat);
    }

    function buildHeavyBag() {
      // The bag swings from an anchor pivot at (0, 3.8, 0)
      bagAssembly = new THREE.Group();
      bagAssembly.position.set(0, 3.8, 0);
      scene.add(bagAssembly);

      // 1. Hanging Steel Chains
      chainGroup = new THREE.Group();
      const chainMat = new THREE.MeshStandardMaterial({ color: 0x9ca3af, metalness: 0.9, roughness: 0.2 });

      for (let c = 0; c < 4; c++) {
        const ang = (c * Math.PI) / 2;
        const radius = 0.9;
        const targetX = Math.cos(ang) * radius;
        const targetZ = Math.sin(ang) * radius;

        // Visual chain strand from apex to bag rim
        const chainGeo = new THREE.CylinderGeometry(0.045, 0.045, 1.8, 8);
        const strand = new THREE.Mesh(chainGeo, chainMat);
        strand.position.set(targetX * 0.5, -0.9, targetZ * 0.5);
        strand.quaternion.setFromUnitVectors(
          new THREE.Vector3(0, 1, 0),
          new THREE.Vector3(targetX, -1.8, targetZ).normalize()
        );
        chainGroup.add(strand);
      }
      bagAssembly.add(chainGroup);

      // 2. Heavy Leather Bag Body
      const bagRadius = 1.15;
      const bagHeight = 4.0;
      const bagGeo = new THREE.CylinderGeometry(bagRadius, bagRadius, bagHeight, 32, 24);

      // Sculpt bulging pillow curvature on the cylinder vertices
      const pos = bagGeo.attributes.position;
      for (let i = 0; i < pos.count; i++) {
        const y = pos.getY(i);
        const bulge = Math.sin(((y + bagHeight * 0.5) / bagHeight) * Math.PI) * 0.12;
        pos.setX(i, pos.getX(i) * (1 + bulge));
        pos.setZ(i, pos.getZ(i) * (1 + bulge));
      }
      bagGeo.computeVertexNormals();

      const bagTex = createPunchingBagTexture();
      const bagMat = new THREE.MeshStandardMaterial({
        map: bagTex,
        roughness: 0.45,
        metalness: 0.2
      });

      heavyBagMesh = new THREE.Mesh(bagGeo, bagMat);
      heavyBagMesh.position.set(0, -3.2, 0); // Positioned below chain apex

      // Thick Anime Inked Outline
      const bagOutline = createAnimeOutline(bagGeo, 0.065);
      heavyBagMesh.add(bagOutline);

      bagAssembly.add(heavyBagMesh);
    }

    function renderSpeedLines() {
      if (!speedlineCtx) return;
      speedlineCtx.clearRect(0, 0, speedlineCanvas.width, speedlineCanvas.height);

      if (speedlineIntensity <= 0.02) return;

      const cx = speedlineCanvas.width * 0.5;
      const cy = speedlineCanvas.height * 0.5;
      const linesCount = Math.floor(speedlineIntensity * 48);

      speedlineCtx.save();
      speedlineCtx.strokeStyle = `rgba(255, 255, 255, ${speedlineIntensity * 0.45})`;
      speedlineCtx.lineWidth = 3 + speedlineIntensity * 4;

      for (let i = 0; i < linesCount; i++) {
        const ang = Math.random() * Math.PI * 2;
        const outerR = Math.max(cx, cy) * 1.2;
        const innerR = (1 - speedlineIntensity * 0.5) * 220 + Math.random() * 80;

        speedlineCtx.beginPath();
        speedlineCtx.moveTo(cx + Math.cos(ang) * innerR, cy + Math.sin(ang) * innerR);
        speedlineCtx.lineTo(cx + Math.cos(ang) * outerR, cy + Math.sin(ang) * outerR);
        speedlineCtx.stroke();
      }
      speedlineCtx.restore();
    }

    function deliverPunch() {
      if (isGachaTriggered) return;

      punchHits++;
      AudioEngine.init();

      // Punch tempo ratio (1.0 to 3.5x)
      const heatRatio = Math.min(punchHits / TARGET_KO_HITS, 1.0);
      const speedFactor = 1.0 + heatRatio * 2.5;

      // Sound & Screen Shake
      AudioEngine.playPunch(speedFactor);
      screenShake = Math.min(0.28, 0.04 + heatRatio * 0.22);

      // Swing bag physically backward and slightly to the hit side
      const pushX = (currentHand === "left" ? 0.35 : -0.35) * (0.8 + heatRatio * 0.7);
      const pushZ = -0.65 * (1.0 + heatRatio * 0.9);
      bagVelocity.z += pushZ;
      bagVelocity.x += pushX;

      // Animate Glove Strike forward
      animateGloveJab(currentHand, speedFactor);
      currentHand = currentHand === "left" ? "right" : "left";

      // Hit Spark Comic Effect
      spawnHitSpark(pushX * 0.5, 0.4, -0.4);

      // Speed lines ramp up
      speedlineIntensity = Math.min(1.0, speedlineIntensity + 0.18);

      // Update HUD
      updateComboHud(heatRatio);

      // Check for KO Gacha Trigger
      if (punchHits >= TARGET_KO_HITS) {
        triggerGachaKnockout();
      }
    }

    function animateGloveJab(hand, speed) {
      const glove = hand === "left" ? leftGloveMesh : rightGloveMesh;
      const restPos = hand === "left" ? new THREE.Vector3(-1.4, -1.2, 5.8) : new THREE.Vector3(1.4, -1.2, 5.8);
      const targetPos = new THREE.Vector3((hand === "left" ? -0.4 : 0.4), 0.2, 2.2);

      const startTime = performance.now();
      const duration = Math.max(90, 220 / speed);

      function stepJab(now) {
        const p = (now - startTime) / duration;
        if (p < 0.4) {
          // Punch forward
          const f = p / 0.4;
          glove.position.lerpVectors(restPos, targetPos, f);
        } else if (p < 1.0) {
          // Recoil back
          const f = (p - 0.4) / 0.6;
          glove.position.lerpVectors(targetPos, restPos, f);
        } else {
          glove.position.copy(restPos);
          return;
        }
        requestAnimationFrame(stepJab);
      }
      requestAnimationFrame(stepJab);
    }

        const easeOut = 1 - Math.pow(1 - progress, 3);

        cardMeshStack.forEach((card, idx) => {
          const delay = idx * 0.09;
          const cardProg = Math.max(0, Math.min((progress - delay) / (1 - delay), 1.0));
          const cardEase = 1 - Math.pow(1 - cardProg, 3);

          // Fly outward into cinematic fan
          card.position.lerpVectors(new THREE.Vector3(0, -0.4, 0), card.userData.fanPos, cardEase);
          card.rotation.x = THREE.MathUtils.lerp(0.8, card.userData.fanRot.x, cardEase);
          card.rotation.y = THREE.MathUtils.lerp(Math.PI, card.userData.fanRot.y, cardEase);
          card.rotation.z = THREE.MathUtils.lerp(0, card.userData.fanRot.z, cardEase);

          const s = Math.min(1.0, cardEase * 1.0);
          card.scale.set(s, s, s);
        });

        if (progress < 1.0) {
          requestAnimationFrame(animateCards);
        } else {
          // Cards locked in position, enable inspection
          document.getElementById("trainAgainBtn").classList.remove("hidden");
          document.getElementById("cardInspectNotice").classList.remove("hidden");
        }
      }
      requestAnimationFrame(animateCards);
    }

    /* =========================================================
       USER INTERACTION: MASH, AUTO-RUSH, RAYCAST & MODAL
       ========================================================= */
    const raycaster = new THREE.Raycaster();
    const mouseVector = new THREE.Vector2();

    function bindInteractions() {
      const container = document.getElementById("webgl-container");

      // Tap anywhere on scene to punch bag (or inspect cards if revealed)
      container.addEventListener("pointerdown", (e) => {
        const rect = container.getBoundingClientRect();
        mouseVector.x = ((e.clientX - rect.left) / container.clientWidth) * 2 - 1;
        mouseVector.y = -((e.clientY - rect.top) / container.clientHeight) * 2 + 1;

        raycaster.setFromCamera(mouseVector, camera);

        if (!isGachaTriggered) {
          // Punch bag!
          deliverPunch();
        } else {
          // Check if user clicked a revealed card
          const cardHits = raycaster.intersectObjects(cardMeshStack, true);
          if (cardHits.length > 0) {
            let rootCard = cardHits[0].object;
            while (rootCard.parent && rootCard.parent !== scene) {
              rootCard = rootCard.parent;
            }
            if (rootCard.userData && rootCard.userData.data) {
              openCardModal(rootCard.userData.data);
            }
          }
        }
      });

      // Tap Punch Button
      document.getElementById("tapPunchBtn").addEventListener("click", deliverPunch);

      // Auto Rush Button (Punches faster and faster automatically)
      const flurryBtn = document.getElementById("flurryBtn");
      flurryBtn.addEventListener("click", () => {
        if (autoRushInterval) {
          clearInterval(autoRushInterval);
          autoRushInterval = null;
          return;
        }

        let speedMs = 280;
        function runAutoPunch() {
          if (isGachaTriggered) return;
          deliverPunch();
          speedMs = Math.max(65, speedMs * 0.88); // Accelerate!
          autoRushInterval = setTimeout(runAutoPunch, speedMs);
        }
        runAutoPunch();
      });

      // Sound Toggle
      document.getElementById("soundToggleBtn").addEventListener("click", () => {
        AudioEngine.enabled = !AudioEngine.enabled;
        document.getElementById("soundIcon").innerText = AudioEngine.enabled ? "🔊" : "🔇";
      });

      // Train Again / Reset Button
      document.getElementById("trainAgainBtn").addEventListener("click", resetGym);

      // Modal Controls
      document.getElementById("closeModalBtn").addEventListener("click", closeCardModal);
      document.getElementById("cardInspectModal").addEventListener("click", (e) => {
        if (e.target.id === "cardInspectModal") closeCardModal();
      });
      document.getElementById("flipCardBtn").addEventListener("click", flipModalCard);

      // 3D Card Tilt Inspection
      const tiltFrame = document.getElementById("cardTiltFrame");
      const holoOverlay = document.getElementById("holoSheenOverlay");

      tiltFrame.addEventListener("mousemove", (e) => {
        const rect = tiltFrame.getBoundingClientRect();
        const x = (e.clientX - rect.left) / rect.width - 0.5;
        const y = (e.clientY - rect.top) / rect.height - 0.5;

        tiltFrame.style.transform = `perspective(800px) rotateY(${x * 24}deg) rotateX(${-y * 24}deg) scale3d(1.05, 1.05, 1.05)`;

        if (activeCardModalData && activeCardModalData.isFoil) {
          holoOverlay.style.opacity = "0.75";
          holoOverlay.style.background = `linear-gradient(${x * 120 + 90}deg, rgba(255,0,128,0.4), rgba(0,255,255,0.4), rgba(255,255,0,0.4))`;
        }
      });

      tiltFrame.addEventListener("mouseleave", () => {
        tiltFrame.style.transform = `perspective(800px) rotateY(0deg) rotateX(0deg) scale3d(1, 1, 1)`;
        holoOverlay.style.opacity = "0";
      });
    }

    function onWindowResize() {
      const container = document.getElementById("webgl-container");

    /* =========================================================
       MAIN RENDER LOOP: PENDULUM SWING & HIT SPARK PHYSICS
       ========================================================= */
    const clock = new THREE.Clock();

    function animateLoop() {
      requestAnimationFrame(animateLoop);

      const delta = clock.getDelta();
      const time = clock.getElapsedTime();

      // 1. Heavy Bag Pendulum Spring Physics
      if (bagAssembly) {
        // Spring forces pulling back to vertical equilibrium
        const springK = 14.0;
        const damping = 0.94;

        bagVelocity.x += -springK * bagAngle.x * delta;
        bagVelocity.z += -springK * bagAngle.z * delta;

        bagVelocity.x *= damping;
        bagVelocity.z *= damping;

        bagAngle.x += bagVelocity.x * delta;
        bagAngle.z += bagVelocity.z * delta;

        // Apply rotation around top ceiling pivot
        bagAssembly.rotation.x = bagAngle.z;
        bagAssembly.rotation.z = -bagAngle.x;
        bagAssembly.rotation.y = Math.sin(time * 0.8) * 0.05; // Idle ambient turn
      }

      // 2. Camera Screen Shake
      if (screenShake > 0.001) {
        camera.position.x = (Math.random() - 0.5) * screenShake;
        camera.position.y = 0.4 + (Math.random() - 0.5) * screenShake;
        screenShake *= 0.88;
      } else {
        camera.position.x = 0;
        camera.position.y = 0.4;
      }

      // 3. Floating Idle Hover for Revealed Cards
      if (isGachaTriggered) {
        cardMeshStack.forEach((card, i) => {
          card.position.y = card.userData.fanPos.y + Math.sin(time * 2.2 + i * 0.7) * 0.08;
          if (card.userData.data.isFoil) {
            card.rotation.y = card.userData.fanRot.y + Math.sin(time * 3.0) * 0.07;
          }
        });
      }

      // 4. Hit Sparks Lifecycle
      for (let i = activeSparks.length - 1; i >= 0; i--) {
        const spark = activeSparks[i];
        spark.position.addScaledVector(spark.userData.velocity, delta);
        spark.userData.life -= delta;
        spark.scale.setScalar(Math.max(0.001, spark.userData.life * 3.0));

        if (spark.userData.life <= 0) {
          scene.remove(spark);
          activeSparks.splice(i, 1);
        }
      }

      // 5. Anime Speedlines Decay
      if (!isGachaTriggered) {
        speedlineIntensity = Math.max(0, speedlineIntensity - delta * 0.45);
      }
      renderSpeedLines();

      renderer.render(scene, camera);
    }
